use esp_hal::{
    clock::ClockControl,
    peripherals::Peripherals,
    prelude::*,
    mcpwm::{
        operator::PwmPinConfig,
        timer::PwmWorkingMode,
        PeripheralClockConfig,
        McPwm,
    },
};

// Assuming clocks and peripherals are already initialized
let clock_cfg = PeripheralClockConfig::with_frequency(40.MHz()).unwrap();
let mut mcpwm = McPwm::new(peripherals.MCPWM0, clock_cfg);

// Connect operator to timer
mcpwm.operator0.set_timer(&mcpwm.timer0);

// Assign pin
let mut pwm_pin = mcpwm.operator0.with_pin_a(pin, PwmPinConfig::UP_ACTIVE_HIGH);

// Start timer at 20 kHz
let timer_clock_cfg = clock_cfg.timer_clock_with_frequency(99, PwmWorkingMode::Increase, 20.kHz()).unwrap();
mcpwm.timer0.start(timer_clock_cfg);

// Set 50% duty cycle (timestamp 50 out of 100)
pwm_pin.set_timestamp(50);
