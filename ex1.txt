use esp_idf_hal::{
    delay::FreeRtos,
    ledc::{config::TimerConfig, LedcDriver, LedcTimerDriver, Resolution},
    peripherals::Peripherals,
    prelude::*,
};

let peripherals = Peripherals::take().unwrap();

// Configure 50 Hz timer for servo control
let timer_driver = LedcTimerDriver::new(
    peripherals.ledc.timer0,
    &TimerConfig::default()
        .frequency(50.Hz())
        .resolution(Resolution::Bits14),
).unwrap();

// Initialize channel on GPIO7
let mut driver = LedcDriver::new(
    peripherals.ledc.channel0,
    timer_driver,
    peripherals.pins.gpio7,
).unwrap();

// Calculate duty limits for 0-180 degrees
let max_duty = driver.get_max_duty();
let min_limit = max_duty * 25 / 1000; // ~1ms pulse
let max_limit = max_duty * 125 / 1000; // ~2ms pulse

// Sweep from 0 to 180 degrees
for angle in 0..180 {
    driver.set_duty(map(angle, 0, 180, min_limit, max_limit)).unwrap();
    FreeRtos::delay_ms(12);
}

fn map(x: u32, in_min: u32, in_max: u32, out_min: u32, out_max: u32) -> u32 {
    (x - in_min) * (out_max - out_min) / (in_max - in_min) + out_min
}
